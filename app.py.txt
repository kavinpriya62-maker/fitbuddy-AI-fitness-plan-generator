import streamlit as st
import google.generativeai as genai

genai.configure(api_key="YOUR_GEMINI_API_KEY")
model = genai.GenerativeModel("gemini-1.5-flash")

st.set_page_config(page_title="FitBuddy", page_icon="💪")
st.title("💪 FitBuddy - Your AI Fitness Partner")
st.caption("Indian Diet + Home Workout - No Gym Needed!")

col1, col2, col3 = st.columns(3)
with col1:
    age = st.number_input("Age", 15, 60, 20)
with col2:
    weight = st.number_input("Weight kg", 30, 150, 65)
with col3:
    height = st.number_input("Height cm", 120, 200, 170)

goal = st.selectbox("Goal Enna?", ["Weight Loss", "Muscle Gain", "Weight Gain", "Stay Fit & Healthy"])
diet_type = st.selectbox("Diet Preference", ["Veg", "Non-Veg", "Eggetarian"])
workout_place = st.radio("Workout Enga?", ["Home", "Gym"])
medical = st.text_input("Any health issue? (Ex: None, Asthma)")

if st.button("🔥 Generate My Plan"):
    with st.spinner("Plan create panrom..."):
        prompt = f"""
        Create fitness plan for: {age} years, {weight}kg, {height}cm, Goal {goal}, Diet {diet_type}, Workout at {workout_place}, Health: {medical}.
        Give: 
        1. BMI calculation
        2. 7-day Indian diet chart (Breakfast/Lunch/Dinner with calories)
        3. 7-day workout plan (30 min daily, no equipments if Home)
        4. Water & Sleep tips
        Language: Simple Tamil + English mix, motivating.
        """
        res = model.generate_content(prompt)
        st.balloons()
        st.write(res.text)

st.sidebar.markdown("NASSCOM FSP - Generative AI Project")
